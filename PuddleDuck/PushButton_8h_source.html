179 virtual void animate() override
180 {
181 unsigned long now = millis(); // current (relative) time in msecs.
182 unsigned long waitTime = (now - fStartTime);
183
184 bool activeLevel = (digitalRead(fPin) == fButtonPressed);
185 switch (fState)
186 {
187 case kPB_INIT:
188 // waiting for level to become active.
189 if (activeLevel)
190 {
191 newState(kPB_DOWN);
192 fStartTime = now; // remember starting time
193 fNClicks = 0;
194 }
195 break;
196
197 case kPB_DOWN:
198 // waiting for level to become inactive.
199
200 if ((!activeLevel) && (waitTime < fDebounceTicks))
201 {
202 // button was released to quickly so I assume some bouncing.
203 newState(fLastState);
204 }
205 else if (!activeLevel)
206 {
207 newState(kPB_UP);
208 fStartTime = now; // remember starting time
209 }
210 else if ((activeLevel) && (waitTime > fPressTicks))
211 {
212 if (fLongPressStartFunc)
213 fLongPressStartFunc();
214 if (fParamLongPressStartFunc)
215 fParamLongPressStartFunc(fLongPressStartFuncParam);
216 newState(kPB_PRESS);
217 }
218 break;
219
220 case kPB_UP:
221 // level is inactive
222 if ((activeLevel) && (waitTime < fDebounceTicks))
223 {
224 // button was pressed to quickly so I assume some bouncing.
225 newState(fLastState); // go back
226 }
227 else if (waitTime >= fDebounceTicks)
228 {
229 // count as a short button down
230 fNClicks++;
231 newState(kPB_COUNT);
232 } // if
233 break;
234
235 case kPB_COUNT:
236 // dobounce time is over, count clicks
237 if (activeLevel)
238 {
239 // button is down again
240 newState(kPB_DOWN);
241 fStartTime = now; // remember starting time
242 }
243 else if ((waitTime > fClickTicks) || (fNClicks == fMaxClicks))
244 {
245 // now we know how many clicks have been made.
246 if (fNClicks == 1)
247 {
248 // this was 1 click only.
249 if (fClickFunc)
250 fClickFunc();
251 if (fParamClickFunc)
252 fParamClickFunc(fClickFuncParam);
253 }
254 else if (fNClicks == 2)
255 {
256 // this was a 2 click sequence.
257 if (fDoubleClickFunc)
258 fDoubleClickFunc();
259 if (fParamDoubleClickFunc)
260 fParamDoubleClickFunc(fDoubleClickFuncParam);
261 }
262 else
263 {
264 // this was a multi click sequence.
265 if (fMultiClickFunc)
266 fMultiClickFunc();
267 if (fParamMultiClickFunc)
268 fParamMultiClickFunc(fMultiClickFuncParam);
269 }
270 reset();
271 }
272 break;
273
274 case kPB_PRESS:
275 // waiting for menu pin being release after long press.
276 if (!activeLevel)
277 {
278 newState(kPB_PRESSEND);
279 fStartTime = now;
280 }
281 else
282 {
283 // still the button is pressed
284 if (fDuringLongPressFunc)
285 fDuringLongPressFunc();
286 if (fParamDuringLongPressFunc)
287 fParamDuringLongPressFunc(fDuringLongPressFuncParam);
288 }
289 break;
290
291 case kPB_PRESSEND:
292 // button was released.
293 if ((activeLevel) && (waitTime < fDebounceTicks))
294 {
295 // button was released to quickly so I assume some bouncing.
296 newState(fLastState); // go back
297 }
298 else if (waitTime >= fDebounceTicks)
299 {
300 if (fLongPressStopFunc)
301 fLongPressStopFunc();
302 if (fParamLongPressStopFunc)
303 fParamLongPressStopFunc(fLongPressStopFuncParam);
304 reset();
305 }
306 break;
307
308 default:
309 // unknown state detected -> reset state machine
310 newState(kPB_INIT);
311 break;
312 }
313 }
