203 // Modeled after shifted quadrant II of unit circle

209 // Modeled after the piecewise circular function
210 // y = (1/2)(1 - sqrt(1 - 4x^2)) ; [0, 0.5)
211 // y = (1/2)(sqrt(-(2x - 3)*(2x - 1)) + 1) ; [0.5, 1]

224 // Modeled after the exponential function y = 2^(10(x - 1))
225 static double ExponentialEaseIn(double p)
226 {
227 return (p == 0.0) ? p : pow(2, 10 * (p - 1));
228 }

230 // Modeled after the exponential function y = -2^(-10x) + 1
231 static double ExponentialEaseOut(double p)
232 {
233 return (p == 1.0) ? p : 1 - pow(2, -10 * p);
234 }

236 // Modeled after the piecewise exponential
237 // y = (1/2)2^(10(2x - 1)) ; [0,0.5)
238 // y = -(1/2)*2^(-10(2x - 1))) + 1 ; [0.5,1]
239 static double ExponentialEaseInOut(double p)
240 {
241 if (p == 0.0 || p == 1.0)
242 {
243 return p;

246 {
247 return 0.5 * pow(2, (20 * p) - 10);
248 }
249 return -0.5 * pow(2, (-20 * p) + 10) + 1;

264 // Modeled after the piecewise exponentially-damped sine wave:
265 // y = (1/2)*sin(13pi/2*(2*x))*pow(2, 10 * ((2*x) - 1)) ; [0,0.5)
266 // y = (1/2)*(sin(-13pi/2*((2x-1)+1))*pow(2,-10(2*x-1)) + 2) ; [0.5, 1]
267 static double ElasticEaseInOut(double p)
268 {
269 if (p < 0.5)

272 }
273 else
274 {
275 return 0.5 * (sin(-13 * M_PI_2 * ((2 * p - 1) + 1)) * pow(2, -10 * (2 * p - 1)) + 2);
276 }

279 // Modeled after the overshooting cubic y = x^3-x*sin(x*pi)
280 static double BackEaseIn(double p)
281 {
282 return p * p * p - p * sin(p * M_PI);
283 }

285 // Modeled after overshooting cubic y = 1-((1-x)^3-(1-x)*sin((1-x)*pi))
286 static double BackEaseOut(double p)
287 {
288 double f = (1 - p);
289 return 1 - (f * f * f - f * sin(f * M_PI));
290 }
291
292 // Modeled after the piecewise overshooting cubic function:
293 // y = (1/2)*((2x)^3-(2x)*sin(2*x*pi)) ; [0, 0.5)
294 // y = (1/2)*(1-((1-x)^3-(1-x)*sin((1-x)*pi))+1) ; [0.5, 1]
295 static double BackEaseInOut(double p)
296 {
297 if (p < 0.5)
298 {
299 double f = 2 * p;
300 return 0.5 * (f * f * f - f * sin(f * M_PI));
301 }
302 else
303 {
304 double f = (1 - (2*p - 1));
305 return 0.5 * (1 - (f * f * f - f * sin(f * M_PI))) + 0.5;
306 }
307 }

309 static double BounceEaseIn(double p)
310 {
311 return 1 - BounceEaseOut(1 - p);
312 }

314 static double BounceEaseOut(double p)
315 {
316 if (p < 4/11.0)
317 {
318 return (121 * p * p)/16.0;
319 }
320 else if (p < 8/11.0)
321 {
322 return (363/40.0 * p * p) - (99/10.0 * p) + 17/5.0;
323 }
324 else if (p < 9/10.0)
325 {
326 return (4356/361.0 * p * p) - (35442/1805.0 * p) + 16061/1805.0;
327 }
328 return (54/5.0 * p * p) - (513/25.0 * p) + 268/25.0;
329 }

331 static double BounceEaseInOut(double p)
332 {
333 if (p < 0.5)
334 {
335 return 0.5 * BounceEaseIn(p*2);
336 }
337 else
338 {
339 return 0.5 * BounceEaseOut(p * 2 - 1) + 0.5;
