96 ssize_t read(void* buffer, size_t bufferSize, uint32_t* id = nullptr) {
97 ssize_t bytesRead = -1;
98 if (isOpen()) {
99 bytesRead = pd::platform::read(fStream, buffer, bufferSize, id);
100 if (bytesRead > 0 && (pd::Log::isVerboseMotor() || fDebug)) {
101 if (bytesRead == 1) {
102 if (!PDLOG_IS_BYTEWISE()) {
103 PDLOG_INFO("[%s:R] ", fName.c_str());
104 }
105 PDLOG_BYTE(((uint8_t*)buffer)[0]);
106 } else {
107 PDLOG_INFO("[%s:R] ", fName.c_str());
108 for (ssize_t i = 0; i < bytesRead; i++) {
109 PDLOG_INFO("%02X ", ((uint8_t*)buffer)[i]);
110 }
111 PDLOG_INFO("\n");
112 }
113 }
114 }
115 return bytesRead;
116 }

118 ssize_t write(const void* buffer, size_t bufferSize, uint32_t id = 0, bool ext = false) {
119 if (isOpen()) {
120 if (bufferSize > 0 && (pd::Log::isVerboseMotor() || fDebug)) {
121 PDLOG_INFO("[%s:W] ", fName.c_str());
122 for (unsigned i = 0; i < bufferSize; i++) {
123 PDLOG_INFO("%02X ", ((uint8_t*)buffer)[i]);
124 }
125 PDLOG_INFO("\n");
126 }
127 return pd::platform::write(fStream, buffer, bufferSize, id, ext);
128 }
129 return -1;
