34 void activateFlashFlight(uint8_t duration, bool enable = true) {
35 if (fDXL != nullptr) {
36 uint8_t data[2] = { duration, static_cast<uint8_t>(enable) };
37 PDLOG_INFO("QuackHead::activateFlashFlight(%d, %d)\r\n", data[0], data[1]);
38 fDXL->write(fID, ADDR_FLASHLIGHT, data, sizeof(data), 250);
39 }
40 }

44 fDXL->write(fID, ADDR_CONTROL_SOUND, &sound, sizeof(sound), 250);
45 }

47 void setVolume(uint8_t volume) {
48 // printf("setVolume\r\n");

51 // printf ("volume command sent\r\n");

83 virtual bool getEvent(Event &event) override {
84 if (fHasOrientation) {
85 event.type = getType();
86 event.sensor_id = fID;
87 event.timestamp = pd::platform::currentTimeMillis();
88 event.gyro.x = fOrientation.x;
89 event.gyro.y = fOrientation.y;
90 event.gyro.z = fOrientation.z;
91 event.acceleration.accuracy = Sensor::Accuracy(fOrientation.accuracy);
92 setCurrentEvent(event);
93 return true;
94 }
95 return false;

97
98 void update() {
99 // Fetch orientation from QuackHead
100 if (fDXL != nullptr) {
101 auto bytesRead = fDXL->read(fID, ADDR_CONTROL_ORIENTATION,
102 sizeof(fOrientation), &fOrientation, sizeof(fOrientation));
103 if (bytesRead == -1) {
104 /* Quackhead not responding */
105 } else if (bytesRead == sizeof(fOrientation)) {
106 fLastResponse = pd::platform::currentTimeMillis();
107 fHasOrientation = true;
108 }
109 }
110 }
111
112 void disableIfNotResponding() {
113 if (!isResponding() && fDXL != nullptr) {
114 PDLOG_INFO("[QUACK] Disabled due to lack of response\n");
115 setDynamixelBus(nullptr);
116 }
117 }
118
119 inline uint32_t timeSinceLastResponse() const {
120 if (fLastResponse) {
121 return pd::platform::currentTimeMillis() - fLastResponse;
122 }
123 return ~0;
124 }
125
126 inline bool isResponding() const {
127 return (timeSinceLastResponse() < 500);
128 }
129
130 virtual void setBus(pd::Bus* bus) override {
131 Sensor::setBus(bus);

134 }
135 }

137protected:

142 static constexpr uint16_t ADDR_CONTROL_VOLUME = 60;
143 DynamixelBus* fDXL = nullptr;
144 bool fHasOrientation = false;
145 uint64_t fLastResponse = 0;
146 Orientation fOrientation = {};
