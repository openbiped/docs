27 bool read() {
28 fHaveData = false;
29 if (fBus == nullptr)
30 return false;
31
32 // Our packet begins with a 4-byte header.
33 const size_t headerSize = sizeof(fData.header);
34
35 // Look for the header signature by shifting in one byte at a time.
36 while (fBus->available()) {
37 uint8_t* header = fData.header;
38 memmove(header, header + 1, headerSize - 1);
39 if (fBus->read(header + headerSize - 1, 1) == 1) {
40 // Check for the fixed header: FE EF 01 7C
41 if (header[0] == 0xFE && header[1] == 0xEF &&
42 header[2] == 0x01 && header[3] == 0x7C)
43 {
44 // Read the rest of the packet. The full packet is 60 bytes.
45 const ssize_t packetRemaining = sizeof(fData) - headerSize;
46 uint8_t* packet = fData.bytes + headerSize; // pointer after header
47
48 if (fBus->read(packet, packetRemaining) == packetRemaining) {
49 // Compute CRC over the first 58 bytes.
50 uint16_t computedCRC = pd::crc(fData.bytes, sizeof(fData.bytes) - sizeof(fData.crc));
51
52 // TODO figure out correct CRC algo
53 computedCRC = fData.crc;
54 if (computedCRC != fData.crc) {
55 PDLOG_INFO("CRC error: computed 0x%04X != received 0x%04X\n",
56 computedCRC, fData.crc);
57 } else {
58 if (pd::Log::isVerboseBMS()) {
59 PDLOG_INFO("Valid BMS packet received:\n");
60
61 PDLOG_INFO("Cell Voltages (V):\n");
62 for (int i = 0; i < 6; ++i) {
63 PDLOG_INFO(" Cell %d: %.3f V\n", i + 1, getCellVoltage(i));
64 }
65 PDLOG_INFO("Total Voltage: %.3f V\n", getPackVoltage());
66
67 PDLOG_INFO("Temperatures (°C):\n");
68 for (int i = 0; i < 4; ++i) {
69 PDLOG_INFO(" Sensor %d: %.1f°C\n", i + 1, getTemperature(i));
70 }
71
72 // Unknown fields.
73
74 // PDLOG_INFO("Status Flags: 0x%04X\n", fData.statusFlags);
75
76 // PDLOG_INFO("ADC Readings:\n");
77 // for (int i = 0; i < 10; i++) {
78 // PDLOG_INFO(" ADC[%d]: %u\n", i, fData.adcReadings[i]);
79 // }
80
81 // PDLOG_INFO("Dynamic Measurement: %u\n", fData.dynamicMeasurement);
82 // PDLOG_INFO("Packet CRC: 0x%04X (verified)\n", fData.crc);
83
84 PDLOG_INFO("Pack Voltage Field: %.3f V\n", getVoltageField());
85 PDLOG_INFO("State of Charge (SOC): %u%% %02Xh\n", getStateOfCharge(), fData.stateOfCharge);
86 PDLOG_INFO("Battery Current: %.3f A\n", getCurrent());
87 }
88 // Print out the full contents of the packet in Hex format.
89 if (pd::Log::isVerboseBMS()) {
90 PDLOG_INFO("Full packet contents:\n");
91 for (size_t i = 0; i < sizeof(fData); ++i) {
92 if (i % 16 == 0) {
93 PDLOG_INFO("\n");
94 }
95 PDLOG_INFO("%02X ", fData.bytes[i]);
96 }
97 PDLOG_INFO("\n");
98 }
99 fHaveData = true;
100 return true;
101 }
102 }
103 // Clear the header buffer before looking for the next packet.
104 memset(fData.header, 0, headerSize);
105 }
106 }
107 }
108 return false;
109 }

131 // Added accessor methods for individual cell voltages and temperatures
132 double getCellVoltage(int index) const {
133 switch (index) {
134 case 0: return fData.measurements.cellVoltage0 / 1000.0;
135 case 1: return fData.measurements.cellVoltage1 / 1000.0;
136 case 2: return fData.measurements.cellVoltage2 / 1000.0;
137 case 3: return fData.measurements.cellVoltage3 / 1000.0;
138 case 4: return fData.measurements.cellVoltage4 / 1000.0;
139 case 5: return fData.measurements.cellVoltage5 / 1000.0;
140 default: return 0;

154 double getPackVoltage() const {
155 double total = 0;
156 for (int i = 0; i < 6; ++i) {
157 total += getCellVoltage(i);
158 }
159 return total;
160 }

162 double getVoltageField() const {

166 uint8_t getStateOfCharge() const {
167 uint8_t soc = static_cast<uint8_t>(fData.stateOfCharge);
168 if (soc <= 100) {
169 fLastValidSOC = soc;

171 return fLastValidSOC;
