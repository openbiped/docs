108 /*
109 * Proportional Gain (kp):
110 * - Definition: The proportional gain kp determines the response of the controller to the current error.
111 * It is a scaling factor applied to the error value.
112 * - Effect:
113 * - Increasing kp: This results in a stronger response to the error, making the system react more quickly
114 * to discrepancies between the desired and actual positions (or other controlled variables). However, if kp
115 * is too high, it can cause the system to become overly aggressive, leading to oscillations or instability.
116 * - Decreasing kp: This results in a weaker response to the error, making the system slower to react. While this
117 * can reduce oscillations and improve stability, it may also result in sluggish performance and a slower return
118 * to the desired position.
119 */
120 void setKP(double kp) {
121 fKP = kp;
122 fChanged = true;
123 }
124
125 inline double getKP() {
126 return fKP;
127 }

129 /*
130 * Derivative Gain (kd):
131 * - Definition: The derivative gain kd determines the response of the controller to the rate of change of the error.
132 * It is a scaling factor applied to the derivative (rate of change) of the error.
133 * - Effect:
134 * - Increasing kd: This results in a stronger damping effect, helping to smooth out the response and reduce overshoot
135 * and oscillations. Essentially, it predicts future error based on the current rate of change and adjusts the control
136 * effort accordingly.
137 * - Decreasing kd: This results in a weaker damping effect, which can lead to increased oscillations and overshoot if
138 * the proportional gain kp is high. However, if kd is too low, the system might not sufficiently counteract rapid changes
139 * in the error, leading to a less smooth response.
140 */
141 void setKD(double kd) {
142 fKD = kd;
143 fChanged = true;
144 }

150 /* Integral Gain (ki):
151 * - Definition: The integral gain ki determines the response of the controller to the accumulated error over time.
152 * It is a scaling factor applied to the integral (sum) of the error.
153 * - Effect:
154 * - Increasing ki: This results in a stronger response to the accumulated error, helping to eliminate steady-state
155 * error and improve the system's ability to track the desired position. However, if ki is too high, it can lead to
156 * instability and oscillations.
157 * - Decreasing ki: This results in a weaker response to the accumulated error, which can help to reduce overshoot and
158 * oscillations. However, if ki is too low, the system may not be able to eliminate steady-state error effectively.
159 */

207 /*
208 * mapDegreesToPosition(double degrees) const:
209 * - Description: Maps motor degrees to a normalized position value in the range [-1.0, 1.0].
210 * - Parameters:
211 * - degrees: The motor's angular position in degrees.
212 * - Returns: A double representing the normalized position in the range [-1.0, 1.0].
213 * - Usage: Use this function to convert an angular position in degrees to a normalized
214 * position suitable for control algorithms or user interfaces.
215 */
216 double mapDegreesToPosition(double degrees) const {
217 constexpr double TOLERANCE = 10.0; // Out-of-range tolerance
218 constexpr double EPSILON = 1e-9; // Floating compare tolerance
219
220 // Retrieve our angles:
221 // - minDeg => maps to -1
222 // - maxDeg => maps to +1
223 // - zeroDeg => maps to 0 (could be equal to minDeg or maxDeg or anywhere in between)
224 double zeroDeg = fZeroDegrees;
225 double minDeg = getLowerRange();
226 double maxDeg = getUpperRange();
227
228 // Out-of-range check: find numeric lo/hi
229 double lo = std::min(minDeg, maxDeg);
230 double hi = std::max(minDeg, maxDeg);
231
232 // PDLOG_INFO("degrees: %f [%f:%f:%f]\n", degrees, zeroDeg, minDeg, maxDeg);
233
234 if (degrees < lo - TOLERANCE || degrees > hi + TOLERANCE) {
235 char id_buf[30];snprintf(id_buf, sizeof(id_buf), "TOL-%s", fName);
236 char msg_buf[110];snprintf(msg_buf, sizeof(msg_buf), "%s Degrees %f is outside allowed [%f..%f] ± tolerance\n", fName, degrees, lo, hi);
237 PDLOG_INFO_THROTTLED(id_buf, msg_buf);
238 return NAN;
239 }
240
241 // Check for edge cases:
242 bool zeroEqMin = (std::fabs(zeroDeg - minDeg) < EPSILON);
243 bool zeroEqMax = (std::fabs(zeroDeg - maxDeg) < EPSILON);
244
245 double position = 0.0;
246
247 //--------------------------------------------------------------------------
248 // CASE A: zeroDeg == minDeg => We have [0..+1] only
249 //--------------------------------------------------------------------------
250 if (zeroEqMin && !zeroEqMax) {
251 // e.g. minDeg=100, zeroDeg=100, maxDeg=140 => position in [0..+1].
252 double denom = (maxDeg - minDeg); // should be > 0 or < 0 if reversed
253 if (std::fabs(denom) < EPSILON) {
254 // minDeg == maxDeg => no range at all
255 PDLOG_INFO("minDeg == maxDeg => no range.\n");
256 return 0.0; // Or NAN, or whatever you prefer
257 }
258
259 // Map [minDeg..maxDeg] => [0..+1]
260 // pos = (degrees - minDeg) / (maxDeg - minDeg)
261 position = (degrees - minDeg) / denom;
262
263 // clamp to [0..+1]
264 if (position < 0.0) position = 0.0;
265 if (position > 1.0) position = 1.0;
266 }
267 //--------------------------------------------------------------------------
268 // CASE B: zeroDeg == maxDeg => We have [-1..0] only
269 //--------------------------------------------------------------------------
270 else if (zeroEqMax && !zeroEqMin) {
271 // e.g. minDeg=100, maxDeg=140, zeroDeg=140 => position in [-1..0].
272 double denom = (maxDeg - minDeg);
273 if (std::fabs(denom) < EPSILON) {
274 // minDeg == maxDeg => no range
275 PDLOG_INFO("minDeg == maxDeg => no range.\n");
276 return 0.0;
277 }
278
279 // Map [minDeg..maxDeg] => [-1..0]
280 // pos = - (maxDeg - degrees) / (maxDeg - minDeg)
281 // so that at degrees=minDeg => pos=-1
282 // degrees=maxDeg => pos= 0
283 position = - (maxDeg - degrees) / denom;
284
285 // clamp to [-1..0]
286 if (position < -1.0) position = -1.0;
287 if (position > 0.0) position = 0.0;
288 }
289 //--------------------------------------------------------------------------
290 // CASE C: zeroDeg is distinct from both minDeg and maxDeg => full [-1..+1]
291 //--------------------------------------------------------------------------
292 else {
293 if (degrees <= zeroDeg) {
294 // Negative side: map [minDeg..zeroDeg] => [-1..0]
295 double denom = (zeroDeg - minDeg);
296 if (std::fabs(denom) < EPSILON) {
297 // If zeroDeg == minDeg, we've fallen into an edge case, but let's clamp
298 PDLOG_INFO("zeroDeg==minDeg in full-range block => clamp to 0.\n");
299 position = 0.0;
300 } else {
301 position = - (zeroDeg - degrees) / denom; // from -1 at minDeg to 0 at zeroDeg
302 }
303 } else {
304 // Positive side: map [zeroDeg..maxDeg] => [0..+1]
305 double denom = (maxDeg - zeroDeg);
306 if (std::fabs(denom) < EPSILON) {
307 // If zeroDeg == maxDeg, we've fallen into another edge case
308 PDLOG_INFO("zeroDeg==maxDeg in full-range block => clamp to 0.\n");
309 position = 0.0;
310 } else {
311 position = (degrees - zeroDeg) / denom; // from 0 at zeroDeg to +1 at maxDeg
312 }
313 }
314
315 // clamp to [-1..+1]
316 if (position < -1.0) position = -1.0;
317 if (position > 1.0) position = 1.0;
318 }
319
320 // 4) If the final position is out of [-1..+1] by more than TOLERANCE, return NAN
321 // (Optional, depends on how strict you want to be)
322 if (position < -1.0 - TOLERANCE || position > 1.0 + TOLERANCE) {
323 PDLOG_INFO("Position %f is out of [-1..+1] by tolerance.\n", position);
324 return NAN;
325 }
326
327 return (isInverted()) ? -position : position;
328 }

330 /*
331 * mapPositionToDegrees(double pos) const:
332 * - Description: Maps a normalized position value in the range [-1.0, 1.0] to motor degrees.
333 * - Parameters:
334 * - pos: The normalized position value in the range [-1.0, 1.0].
335 * - Returns: A double representing the motor's angular position in degrees.
336 * - Usage: Use this function to convert a normalized position value to an angular position in
337 * degrees for motor control or visualization.
338 */
339 double mapPositionToDegrees(double pos) const {
340 // Handle inversion if needed
341 if (isInverted()) {
342 pos = -pos;
343 }
344
345 // Read the configured angles:
346 // lowerDeg => angle for position = -1
347 // upperDeg => angle for position = +1
348 // zeroDeg => angle for position = 0
349 double lowerDeg = getLowerRange();
350 double upperDeg = getUpperRange();
351 double zeroDeg = fZeroDegrees;
352
353 // Piecewise calculation:
354 // If pos < 0 => map [-1..0] to [lowerDeg..zeroDeg].
355 // If pos >=0 => map [ 0..+1] to [zeroDeg..upperDeg].
356 double degrees;
357 if (pos < 0.0)
358 {
359 // Mapping formula:
360 // pos = -1 => degrees = lowerDeg
361 // pos = 0 => degrees = zeroDeg
362 // So degrees = zeroDeg + (pos * (zeroDeg - lowerDeg))
363 //
364 // Explanation:
365 // - For pos=-1, the term (pos*(zeroDeg - lowerDeg)) = -1*(zeroDeg - lowerDeg) = lowerDeg - zeroDeg,
366 // so degrees = zeroDeg + (lowerDeg - zeroDeg) = lowerDeg
367 degrees = zeroDeg + pos * (zeroDeg - lowerDeg);
368 }
369 else
370 {
371 // Mapping formula:
372 // pos = 0 => degrees = zeroDeg
373 // pos = +1 => degrees = upperDeg
374 //
375 // So degrees = zeroDeg + (pos * (upperDeg - zeroDeg))
376 degrees = zeroDeg + pos * (upperDeg - zeroDeg);
377 }
378
379 // Clamp to the valid mechanical range [lo..hi], where lo/hi = min/max of lowerDeg,upperDeg
380 double lo = std::min(lowerDeg, upperDeg);
381 double hi = std::max(lowerDeg, upperDeg);
382 if (degrees < lo) {
383 degrees = lo;
384 }
385 else if (degrees > hi) {
386 degrees = hi;
387 }
388
389 // PDLOG_INFO("map lower=%f upper=%f zero=%f pos=%f => degrees=%f\n", lowerDeg, upperDeg, zeroDeg, pos, degrees);
390 return degrees;
391 }

394 * moveToPosition(double pos, uint32_t moveTime, uint32_t startDelay):
395 * - Description: Initiates a movement to a specified normalized position after a delay.
396 * - Parameters:
397 * - pos: The target position as a normalized value in the range [-1.0, 1.0].
398 * - moveTime: The duration in milliseconds for the movement to complete.
399 * - startDelay: The delay in milliseconds before starting the movement.
400 * - Usage: Use this function to move the motor to a specified normalized position with a specified delay and movement duration.
401 * - Note: This function internally maps the normalized position to motor degrees and calls moveToDegrees.
402 */

407 /*
408 * moveToDegrees(double degrees, uint32_t moveTime, uint32_t startDelay):
409 * - Description: Initiates a movement to a specified angular position in degrees after a delay.
410 * - Parameters:
411 * - degrees: The target angular position in degrees.
412 * - moveTime: The duration in milliseconds for the movement to complete.
413 * - startDelay: The delay in milliseconds before starting the movement.
414 * - Usage: Use this function to move the motor to a specified angular position with a specified delay and movement duration.
415 */
416 void moveToDegrees(double degrees, uint32_t moveTime = 0, uint32_t startDelay = 0) {
417 if (std::isnan(getPosition()))
418 return;
419 double lo = getMinimum();
420 double hi = getMaximum();
421 if (degrees < lo)
422 degrees = lo;
423 if (degrees > hi)
424 degrees = hi;
425
426 // If moving over time check if we're already "close enough" to that target:
427 if (moveTime != 0 && std::fabs(degrees - fDegrees) <= fMovementEpsilon)
428 {
429 // This prevents repeated possible overshoot caused by damping.
430 PDLOG_INFO("%s: Already near target=%.2f, current=%.2f, skipping...\n",
431 fName, degrees, fDegrees);
432 return;
433 }
434
435 fActive = true;
436 double finalPos = mapPositionToDegrees(mapDegreesToPosition(degrees));
437 if (finalPos < lo)
438 finalPos = lo;
439 if (finalPos > hi)
440 finalPos = hi;
441 if (moveTime != 0) {
442 uint64_t timeNow = pd::platform::currentTimeMillis();
443 fStartTime = startDelay + timeNow;
444 fFinishTime = moveTime + fStartTime;
445 fOffTime = fFinishTime;
446 fFinishPos = finalPos;
447 fPosNow = fDegrees;
448 fStartPosition = fPosNow;
449 fDeltaPos = fFinishPos - fPosNow;
450 fLastMoveTime = pd::platform::currentTimeMillis();
451 } else {
452 reset();
453 fPosNow = finalPos;
454 }
455 }

457 void moveToJointOffset(double radians, uint32_t moveTime = 0, uint32_t startTime = 0) {
458 double targetDegrees = 0.0;
459 // Calculate scale factors relative to the calibrated zero.
460 double maxRad = degreesToRadians(std::fabs(getUpperRange() - fZeroDegrees));
461 double minRad = degreesToRadians(std::fabs(getLowerRange() - fZeroDegrees));
462
463 double normOffset = 0.0;
464 if (radians > 0.0) {
465 // Positive offset: use the upper side's scale.
466 if (maxRad > 0.0)
467 normOffset = radians / maxRad;
468 }
469 else if (radians < 0.0) {
470 // Negative offset: use the lower side's scale.
471 if (minRad > 0.0)
472 normOffset = radians / minRad; // (radians is negative, so normOffset will be negative)
473 }
474 // For radians == 0, normOffset remains 0.
475
476 // Convert the normalized offset to an absolute target angle.
477 targetDegrees = mapPositionToDegrees(normOffset);
478
479 // PDLOG_INFO(" minDeg=%.1f maxDeg=%.1f zeroDeg=%.1f\n", getLowerRange(), getUpperRange(), getZeroDegrees());
480 // PDLOG_INFO(" normOffset=%.3f => targetDeg=%.3f\n", normOffset, targetDegrees);
481
482 // Command the move.
483 moveToDegrees(targetDegrees, moveTime, startTime);
484 }

486 void move(uint64_t timeNow) {
487 Easing::Method easing = Easing::get(fEasingMethod);
488 if (fFinishTime != 0) {
489 if (timeNow < fStartTime) {
490 /* wait */
491 } else if (timeNow >= fFinishTime) {
492 fPosNow = fFinishPos;
493 reset();
494 } else if (fLastMoveTime != timeNow) {
495 uint32_t timeSinceLastMove = timeNow - fStartTime;
496 uint32_t denominator = fFinishTime - fStartTime;
497 double fractionChange = easing(double(timeSinceLastMove) / double(denominator));
498 double distanceToMove = fDeltaPos * fractionChange;
499 double newPos = fStartPosition + distanceToMove;
500 if (newPos != fPosNow)
501 {
502 fPosNow = fStartPosition + distanceToMove;
503 double lo = getMinimum();
504 double hi = getMaximum();
505 if (fPosNow < lo) fPosNow = lo;
506 if (fPosNow > hi) fPosNow = hi;
507 fLastMoveTime = timeNow;
508 //PDLOG_INFO(" fPosNow=%f\n", fPosNow);
509 }
510 }
511 } else if (fOffTime != 0 && timeNow >= fOffTime) {
512 fOffTime = 0;
513 }
514 }

680 inline double getJointOffsetForPosition(double pos) {
681 double lowerDeg = getLowerRange();
682 double upperDeg = getUpperRange();
683 double zeroDeg = fZeroDegrees;
684 double minRad = degreesToRadians(std::fabs(lowerDeg - zeroDeg));
685 double maxRad = degreesToRadians(std::fabs(upperDeg - zeroDeg));
686
687 // PDLOG_INFO("pos=%f lowerDeg=%f upperDeg=%f zeroDeg=%f minRad=%f maxRad=%f\n",
688 // pos, lowerDeg, upperDeg, zeroDeg, minRad, maxRad);
689 if (pos < -1.0) pos = -1.0;
690 if (pos > 1.0) pos = 1.0;
691
692 // 5) Piecewise mapping:
693 // - pos in [-1..0]: map to [minRad..0]
694 // formula: angleRad = -pos * minRad
695 // -1 => minRad, 0 => 0
696 // - pos in [0..+1]: map to [0..maxRad]
697 // formula: angleRad = pos * maxRad
698 // 0 => 0, +1 => maxRad
699 double angleRad;
700 if (pos < 0.0) {
701 angleRad = pos * minRad; // e.g. if pos=-1 => angleRad = minRad
702 } else {
703 angleRad = pos * maxRad; // e.g. if pos=+1 => angleRad = maxRad
704 }
705 return angleRad;
706 }

712 /*
713 * getPosition() const:
714 * - Description: Retrieves the current normalized position of the motor in the range [-1.0, 1.0].
715 * - Returns: A double representing the motor's current normalized position.
716 * - Usage: Use this function to get the current normalized position of the motor.
717 */
718 inline double getPosition() {
719 if (!isRangeValid()) {
720 pd::String busName;
721 if (fBus != nullptr)
722 busName = fBus->getName();
723 if (!fRangeError) {
724 PDLOG_ERROR("ERROR ACUTATOR: \"%s.%s\" RANGE NOT SET\n", busName.c_str(), fName);
725 fRangeError = true;
726 }
