65 /*
66 * setTau(float tau):
67 * - Description: Sets the torque (tau) for the motor. The torque represents the rotational force being applied.
68 * - Parameters:
69 * - tau: The desired torque value in Newton-meters (Nm).
70 * - Usage: Use this function to control the amount of rotational force the motor applies.
71 */
72 inline void setTau(float tau) {
73 tau = pd::max(pd::min(tau, 127), -127);
74 int16_t tau_int = int16_t(tau*256);
75 cmd.fTau[0] = uint8_t((tau_int>>0)&0xFF);
76 cmd.fTau[1] = uint8_t((tau_int>>8)&0xFF);
77 }

79 /*
80 * setDQ(float dq):
81 * - Description: Sets the angular velocity (dq) for the motor. The angular velocity is the rate of change of the motor's position.
82 * - Parameters:
83 * - dq: The desired angular velocity in radians per second (rad/s).
84 * - Usage: Use this function to control the speed at which the motor rotates.
85 */
86 inline void setDQ(float dq) {
87 dq = pd::max(pd::min(dq, 255.0), -256.0);
88 int16_t dq_int = int16_t(dq*256);
89 cmd.fDQ[0] = uint8_t((dq_int>>0)&0xFF);
90 cmd.fDQ[1] = uint8_t((dq_int>>8)&0xFF);
91 }
92
93 /*
94 * setQ(float q):
95 * - Description: Sets the position (q) for the motor. The position represents the angular displacement of the motor shaft.
96 * - Parameters:
97 * - q: The desired position in radians * gear shaft ratio.
98 * - Usage: Use this function to control the angular position of the motor shaft.
99 */
100 inline void setQ(float q) {
101 q = pd::max(pd::min(q, 411774), -411774);
102 int32_t q_int = int32_t(q/6.2832*32768.0);
103 cmd.fQ[0] = uint8_t((q_int>>0)&0xFF);
104 cmd.fQ[1] = uint8_t((q_int>>8)&0xFF);
105 cmd.fQ[2] = uint8_t((q_int>>16)&0xFF);
106 cmd.fQ[3] = uint8_t((q_int>>24)&0xFF);
107 }

110 * setQRadians(float q):

112 * - Parameters:

114 * - Usage: Use this function to control the angular position of the motor shaft.

131 /*
132 * Proportional Gain (kp):
133 * - Definition: The proportional gain kp determines the response of the controller to the current error.
134 * It is a scaling factor applied to the error value.
135 * - Effect:
136 * - Increasing kp: This results in a stronger response to the error, making the system react more quickly
137 * to discrepancies between the desired and actual positions (or other controlled variables). However, if kp
138 * is too high, it can cause the system to become overly aggressive, leading to oscillations or instability.
139 * - Decreasing kp: This results in a weaker response to the error, making the system slower to react. While this
140 * can reduce oscillations and improve stability, it may also result in sluggish performance and a slower return
141 * to the desired position.
142 */
143 inline void setKP(float output_kp) {
144 float rotor_kp = (output_kp / (GEAR_RATIO * GEAR_RATIO));
145 int16_t kp_int = int16_t(rotor_kp/25.6*32768);
146 cmd.fKP[0] = uint8_t((kp_int>>0)&0xFF);
147 cmd.fKP[1] = uint8_t((kp_int>>8)&0xFF);
148 }
149
150 /*
151 * Derivative Gain (kd):
152 * - Definition: The derivative gain kd determines the response of the controller to the rate of change of the error.
153 * It is a scaling factor applied to the derivative (rate of change) of the error.
154 * - Effect:
155 * - Increasing kd: This results in a stronger damping effect, helping to smooth out the response and reduce overshoot
156 * and oscillations. Essentially, it predicts future error based on the current rate of change and adjusts the control
157 * effort accordingly.
158 * - Decreasing kd: This results in a weaker damping effect, which can lead to increased oscillations and overshoot if
159 * the proportional gain kp is high. However, if kd is too low, the system might not sufficiently counteract rapid
160 * changes in the error, leading to a less smooth response.
161 */
162 inline void setKD(float output_kd) {
163 float rotor_kd = (output_kd / (GEAR_RATIO * GEAR_RATIO));
164 int16_t kd_int = int16_t(rotor_kd/25.6*32768);
165 cmd.fKD[0] = uint8_t((kd_int>>0)&0xFF);
166 cmd.fKD[1] = uint8_t((kd_int>>8)&0xFF);
167 }

290 bool read(pd::Bus* bus, const CRC& motorCRC) {
291 if (bus != nullptr && bus->read(fBytes, sizeof(fBytes)) == sizeof(fBytes)) {
292 uint16_t crc = motorCRC.crcRead(&cmd, sizeof(cmd));
293 // Drain the read buffer
294 bus->flushUnread();
295 if (fCRC[0] == uint8_t(crc & 0xFF) && fCRC[1] == uint8_t(crc >> 8)) {
296 return true;
297 } else {
298 PDLOG_ERROR("BAD CRC EXPECTED %02X:%02X GOT %02X:%02X\n", fCRC[0], fCRC[1], uint8_t(crc & 0xFF), uint8_t(crc >> 8));
299 for (unsigned i = 0; i < sizeof(fBytes); i++) {
300 PDLOG_ERROR("%02X ", fBytes[i]);
301 }
302 PDLOG_ERROR("\n");
303 }
304 }
305 invalid();
306 return false;

355 void update(Cmd& cmd, Feedback& feedback) {
356 if (fIgnore) {
357 cmd.setInvalid();
358 return;
359 }
360 move(pd::platform::currentTimeMillis());
361 cmd.setMotorID(fMotorID);
362 cmd.setDQ(0);
363 if (fActive) {
364 if (pd::Log::isVerboseMove()) {
365 if (inVelocityMode()) {
366 PDLOG_INFO("[%s] KP=%f KD=%f DQ=%f\n", getName().c_str(), fKP, fKD, fDQ);
367 } else {
368 PDLOG_INFO("[%s] KP=%f KD=%f Q=%f\n", getName().c_str(), fKP, fKD, fPosNow);
369 }
370 }
371 cmd.setFOCMode();
372 cmd.setKP(fKP);
373 cmd.setKD(fKD);
374 cmd.setTau(fTau);
375 if (inVelocityMode()) {
376 cmd.setDQ(fDQ);
377 } else {
378 cmd.setQRadians(degreesToRadians(fPosNow));
379 }
380 } else {
381 //cmd.setFOCMode();
382 cmd.setBrakeMode();
383 cmd.setKP(0.00);
384 cmd.setKD(0.00);
385 cmd.setQRadians(0);
386 cmd.setTau(0.0);
387 }
388 feedback.init();

391 bool update(Feedback& feedback) {
392 if (feedback.getMotorID() != fMotorID) {
393 PDLOG_WARNING("WRONG MOTOR GOT %d EXPECTING %d\n", feedback.getMotorID(), fMotorID);
394 return false;
395 }
396 setFeedback(
397 feedback.getCurrentAngle(),
398 feedback.getDQ(),
399 feedback.getTau(),
400 feedback.getTemperature());
401 if (feedback.getVoltageError()) {
402 PDLOG_ERROR("[%d] VOLTAGE ERROR %s.%s\n", fMotorID, fBus->getName().c_str(), getName().c_str());
403 setError(Error::VoltageError);
404 } else if (feedback.getCurrentError()) {
405 PDLOG_ERROR("[%d] CURRENT ERROR %s.%s\n", fMotorID, fBus->getName().c_str(), getName().c_str());
406 setError(Error::CurrentError);
407 } else if (feedback.getTemperatureError()) {
408 PDLOG_ERROR("[%d] TEMP ERROR %s.%s\n", fMotorID, fBus->getName().c_str(), getName().c_str());
409 setError(Error::TemperatureError);
410 } else if (feedback.getEncoderError()) {
411 PDLOG_ERROR("[%d] ENCODER ERROR %s.%s\n", fMotorID, fBus->getName().c_str(), getName().c_str());
412 setError(Error::EncoderError);
413 } else {
414 clearError();
415 }
416 return true;
417 }

419 bool update() override {

463 return true;

465 if (!fCmd.write(fBus, crc)) {
466 PDLOG_ERROR("FAILED TO WRITE MOTOR COMMAND TO %s\n", fBus->getName().c_str());

468 }

479 fFeedback.getDQ(),
480 fFeedback.getTau(),
481 fFeedback.getTemperature());
482 return true;
483 }

504 // Drain the read buffer

506 feedback->init();
507 if (!cmd->write(fBus, crc)) {
508 PDLOG_ERROR("FAILED TO WRITE MOTOR COMMAND TO %s\n", fBus->getName().c_str());
509 return false;

513 return false;
514 }
515 }
516 return true;
