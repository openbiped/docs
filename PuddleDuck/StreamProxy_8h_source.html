16class StreamProxy {
17public:
18 StreamProxy(String name) :
19 fName(name),
20 fNext(nullptr)
21 {
22 if (*head() == NULL)
23 *head() = this;
24 if (*tail() != NULL)
25 (*tail())->fNext = this;
26 *tail() = this;
27 }
28
29 String getName() const {
30 return fName;
31 }
32
33 static pd::platform::Stream getByName(const char* name) {
34 for (StreamProxy* stream = *head(); stream != NULL; stream = stream->fNext) {
35 if (stream->getName() == name) {
36 return stream;
37 }
38 }
39 return nullptr;
40 }
41
42 virtual void begin(unsigned baud) = 0;
43 virtual void close() {}
44 virtual void setTimeout(int centiseconds) = 0;
45 virtual size_t write(const void* buffer, size_t size, uint32_t id, bool ext) = 0;
46 virtual size_t read(char* buffer, size_t size, uint32_t* id) = 0;
47 virtual void flush() {}
48 virtual bool available() = 0;
49
50 virtual bool setRTS() { return false; }
51 virtual void clearRTS() {}
52
53protected:
54 String fName;
55 StreamProxy* fNext;
56
57private:
58 static StreamProxy** head()
59 {
60 static StreamProxy* sHead;
61 return &sHead;
62 }
63
64 static StreamProxy** tail()
65 {
66 static StreamProxy* sTail;
67 return &sTail;
68 }
69};

228 I2CStreamProxy(String name, TwoWire &wire) :
229 StreamProxy(name),
230 fWire(wire)

232 }

234 void begin(unsigned baud) override {
235 }
236
237 void setTimeout(int centiseconds) override {
238 }
239
240 size_t write(const void* buffer, size_t size, uint32_t id, bool ext) override {
241 if (pd::Log::isVerboseI2C()) {
242 PDLOG_INFO("WRITE I2C[0x%08X] ", id);
243 for (size_t i = 0; i < size; i++) {
244 PDLOG_INFO("%02X ", ((uint8_t*)buffer)[i]);
245 }
246 PDLOG_INFO("\n");
247 }
248 fWire.beginTransmission(uint16_t(id));
249 size_t result = fWire.write((uint8_t*)buffer, size);
250 fWire.endTransmission(uint16_t(id));
251 return result;
252 }
253
254 size_t read(char* buffer, size_t size, uint32_t* id) override {
255 size_t bytesRead = 0;
256 if (id != nullptr) {
257 fWire.requestFrom(uint8_t(*id), uint8_t(size));
258 }
259 while (fWire.available() && bytesRead < size) {
260 int ch = fWire.read();
261 if (ch != -1) {
262 buffer[bytesRead++] = ch;
263 }
264 }
265 if (pd::Log::isVerboseI2C()) {
266 if (id != nullptr) {
267 PDLOG_INFO("READ I2C[0x%08X] ", *id);
268 } else {
269 PDLOG_INFO("READ I2C ", id);
270 }
271 for (size_t i = 0; i < bytesRead; i++) {
272 PDLOG_INFO("%02X ", ((uint8_t*)buffer)[i]);
273 }
274 PDLOG_INFO("\n");
275 }
276 return bytesRead;
277 }
278
279 bool available() override {
280 return fWire.available();
281 }
282
283private:
284 TwoWire &fWire;
285};

350 *id = 0;

367public:

369 using OnDisconnectFunc = void (*)(void);
370
371 SerialUSBState(SerialT& serial) :
372 fSerial(serial)
373 {
374 }
375
376 void onConnect(OnConnectFunc onConnect) {
377 fOnConnect = onConnect;
378 }
379
380 void onDisconnect(OnDisconnectFunc onDisconnect) {
381 fOnDisconnect = onDisconnect;
382 }
383
384 void update() {
385 bool dtr = fSerial.dtr();
386 if (!fConnected) {
387 if (dtr) {
388 fConnected = true;
389 if (fOnConnect) {
390 fOnConnect();
391 }
392 }
393 } else if (!dtr) {
394 fConnected = false;
395 if (fOnDisconnect) {
396 fOnDisconnect();
397 }
398 }
399 }
400
401 bool connected() {
402 return fConnected;
403 }
404
405private:
406 SerialT& fSerial;
407 bool fConnected = false;
408 OnConnectFunc fOnConnect = nullptr;
409 OnDisconnectFunc fOnDisconnect = nullptr;
