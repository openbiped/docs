65 Armature::Pose pose;
66 fArmature.getPose(pose);
67 if (pose != fPoseSample.fPose) {
68 uint64_t now = pd::platform::currentTimeMillis();
69 if (fSamples.size() == 0) {
70 fPoseSample.fElapsed = 0;
71 } else {
72 fPoseSample.fElapsed = now - fTimeStamp;
73 }
74 fPoseSample.fPose = pose;
75 fTimeStamp = now;
76 fSamples.push_back(fPoseSample);
77 if (pd::Log::isVerbose()) {
78 printf("add pose: %f, %f, %f, %f, %f\n", pose.fPositions[0], pose.fPositions[1], pose.fPositions[2], pose.fPositions[3], pose.fPositions[4]);
79 }
