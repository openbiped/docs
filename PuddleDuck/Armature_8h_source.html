31 friend bool almostEqual(const Pose& lhs, const Pose& rhs, double tolerance = CLOSE_ENOUGH) {
32 size_t count = std::min(lhs.fPositions.size(), rhs.fPositions.size());
33 for (size_t i = 0; i < count; i++) {
34 if (std::isnan(lhs.fPositions[i]) && std::isnan(rhs.fPositions[i]))

60 Armature(const pd::config::Armature& armature) {
61 snprintf(fName, sizeof(fName), "%s", armature.name.c_str());
62 for (size_t i = 0; i < armature.joints.size(); i++) {
63 const pd::config::Joint &joint = armature.joints[i];
64 pd::Actuator* actuator = nullptr;
65 switch (joint.type) {
66 case pd::Go1:
67 case pd::Go2:
68 actuator = new pd::motor::unitree::Go(joint.id, joint.type, joint.name);
69 break;
70 case pd::UkiA1:
71 actuator = new pd::motor::unitree::UkiA1(joint.id, joint.name);
72 break;
73 case pd::UkiB1:
74 actuator = new pd::motor::unitree::UkiB1(joint.id, joint.name);
75 break;
76 case pd::CyberGear:
77 actuator = new pd::motor::xiaomi::Cybergear(joint.id, joint.name);
78 break;
79 case pd::Dynamixel:
80 actuator = new pd::motor::robotis::Dynamixel(joint.id, joint.type, joint.name);
81 break;
82 case pd::Servo:
83 actuator = new pd::motor::servo::Analog(joint.id, joint.name);
84 break;
85 case pd::UnknownMotor:
86 default:
87 assert(false);
88 break;
89 }
90 if (actuator != nullptr) {
91 actuator->setStartDegrees(joint.start);
92 actuator->setKP((!std::isnan(joint.kp)) ? joint.kp : joint.DEFAULT_KP);
93 actuator->setKI((!std::isnan(joint.ki)) ? joint.ki : joint.DEFAULT_KI);
94 actuator->setKD((!std::isnan(joint.kd)) ? joint.kd : joint.DEFAULT_KD);
95 actuator->setTau((!std::isnan(joint.tau)) ? joint.tau : joint.DEFAULT_TAU);
96 actuator->setInvert(joint.invert);
97 actuator->setOptional(joint.optional);
98 actuator->setRange(joint.range.value[0], joint.range.value[1]);
99 if (joint.zero != 0)
100 actuator->setZeroDegrees(joint.zero);
101 fActuator.push_back(actuator);
102 }
103 }
104 }

181 bool checkRanges() {
182 bool success = true;

229 bool update() {
230 if (fBus == nullptr) {
231 PDLOG_ERROR("[%s] UNRESOLVED ARMATURE BUS\n", fName);
232 return false;
233 }
234 size_t successCount = 0;
235 for (auto actuator : fActuator) {
236 if (actuator->update()) {
237 successCount++;
238 }
239 }
240 bool success = (fActuator.size() == successCount);
241 for (auto actuator : fActuator) {
242 if (!actuator->shouldIgnore() && !actuator->isResponding()) {
243 pd::Bus* bus = actuator->getBus();
244 PDLOG_ERROR("[%s] MOTOR NOT RESPONDING: %s.%s\n",
245 bus->getName().c_str(),
246 fName,
247 actuator->getName().c_str());
248 bus->setError(true);
249 }
250 }
251 if (pd::Log::isVerbosePosition()) {
252 report();
253 }
254 return success;
255 }
