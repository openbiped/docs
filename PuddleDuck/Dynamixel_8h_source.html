12// From the XH540-W150-R e-Manual:
13// Rated torque: 4.1 Nm @ 24V
14// Rated current: ~2.1 A
15// → Approximate Kt ≈ 4.1 Nm / 2.1 A ≈ 1.95 Nm/A
16// → Or 0.00195 Nm/mA
17
18// From the XC330-T181-R e-Manual:
19// Stall torque: 0.57 Nm @ 2.6 A
20// → Approximate Kt ≈ 0.57 / 2.6 ≈ 0.219 Nm/A
21// → Or 0.000219 Nm/mA
22
23#define TORQUE_CONSTANT_XH540 0.00195
24#define TORQUE_CONSTANT_XC330 0.000219
25#define XH540_W150 (uint16_t)1110
26#define XC330_T181 (uint16_t)1210

49 virtual bool update() override {
50 auto dxl = getDynamixelBus();
51 if (shouldIgnore()) {
52 return true;
53 }
54 move(currentTimeMillis());
55 if (dxl == nullptr) {
56 return false;
57 }
58 if (fActive) {
59 bool needPos = false;
60 if (!fTorqueOn) {
61 fDXL->torqueOn(fMotorID);
62 fTorqueOn = true;
63 needPos = true;
64 }
65 if (pd::Log::isVerboseMove()) {
66 PDLOG_INFO("[%s] KP=%f KI= %f KD=%f %f\n", getName().c_str(), fKP, fKI, fKD, fPosNow);
67 }
68 if (needPos || fDegrees != fPosNow) {
69 dxl->setGoalPosition(fMotorID, fPosNow);
70 }
71 } else if (fTorqueOn) {
72 fDXL->torqueOff(fMotorID);
73 fTorqueOn = false;
74 }
75 // TODO: report present velocity
76 // PDLOG_INFO("[%s] degrees=%f\n", getName().c_str(), dxl->getPresentPosition(fMotorID));
77
78 auto type = dxl->getModelNumber(fMotorID);
79 auto pos = dxl->getPresentPosition(fMotorID);
80 auto vel_rpm = dxl->getPresentVelocity(fMotorID);
81
82 double vel_rad_per_sec = vel_rpm * 2.0 * M_PI / 60.0;
83
84 double current_mA = dxl->getPresentCurrent(fMotorID);
85 double torque_nm = 0.0;
86
87 if (pos != 0 && !dxl->hasError()) {
88 if (type == XH540_W150) {
89 torque_nm = current_mA * TORQUE_CONSTANT_XH540;
90 } else if (type == XC330_T181) {
91 torque_nm = current_mA * TORQUE_CONSTANT_XC330;
92 }
93 setFeedback(pos, vel_rad_per_sec, torque_nm);
94 }
95
96 return !hasError();
97 }
98
99 DynamixelBus* getDynamixelBus() {
100 if (fDXL == nullptr) {
101 if (fBus == nullptr) {
102 PDLOG_ERROR("UNRESOLVED ACTUATOR BUS\n");
103 setIgnore();
104 } else {
105 fDXL = DynamixelBus::getDynamixelBus(fBus);
106 if (fDXL == nullptr) {
107 PDLOG_ERROR("UNRESOLVED DYNAMIXEL BUS: %s\n", fBus->getName().c_str());
108 setIgnore();
109 return nullptr;
110 }
111 fDXL->setPortProtocolVersion(2 /*DXL_PROTOCOL_VERSION*/);
112 if (!fDXL->ping(fMotorID)) {
113 if (isOptional()) {
114 PDLOG_INFO("[%s] DYNAMIXEL id=%d [%s] not responding (optional)\n",
115 fBus->getName().c_str(),
116 fMotorID,
117 getName().c_str());
118 setIgnore();
119 } else {
120 fMissCount++;
121 PDLOG_ERROR("[%s] DYNAMIXEL id=%d [%s] NOT RESPONDING\n",
122 fBus->getName().c_str(),
123 fMotorID,
124 getName().c_str());
125 }
126 } else {
127 if (!fDXL->ledOn(fMotorID)) {
128 PDLOG_INFO("LEDON FAILED\n");
129 }
130 if (!fDXL->torqueOff(fMotorID)) {
131 PDLOG_INFO("TORQUE OFF FAILED\n");
132 }
133 if (!fDXL->setOperatingMode(fMotorID, OP_POSITION)) {
134 PDLOG_INFO("OPERATING MODE FAILED\n");
135 }
136 if (!fDXL->writeControlTableItem(POSITION_P_GAIN, fMotorID, fKP)) {
137 PDLOG_INFO("KP FAILED\n");
138 }
139 if (!fDXL->writeControlTableItem(POSITION_I_GAIN, fMotorID, fKI)) {
140 PDLOG_INFO("KI FAILED\n");
141 }
142 if (!fDXL->writeControlTableItem(POSITION_D_GAIN, fMotorID, fKD)) {
143 PDLOG_INFO("KD FAILED\n");
144 }
145 fPosNow = fDegrees = fDXL->getPresentPosition(fMotorID);
146 }
147 }
148 }
149 return fDXL;
150 }
151
152protected:
153 DynamixelBus* fDXL = nullptr;
154 bool fTorqueOn = false;
155};
