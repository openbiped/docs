17class Robot {
18public:
19 Robot(pd::config::Robot& config) :
20 fConfig(config)
21 {
22 for (size_t i = 0; i < config.bus.size(); i++) {
23 const pd::config::Bus &bus = config.bus[i];
24 auto newbus = new Bus(
25 bus.name,
26 bus.type,
27 bus.adapter,
28 bus.baud,
29 bus.id,
30 bus.readtimeout
31 );
32 newbus->setDebug(bus.debug);
33 fBus.push_back(newbus);
34 }
35
36 double filterAlpha = NAN;
37 if (fConfig.cutoff != 0) {
38 double dt = config.dt();
39 double tau = 1.0 / (2 * M_PI * fConfig.cutoff);
40 filterAlpha = dt / (tau + dt);
41 }
42 for (size_t i = 0; i < config.armature.size(); i++) {
43 const pd::config::Armature &armature = config.armature[i];
44 Armature* arm = new Armature(armature);
45 arm->setBus(getBus(armature.name, armature.bus));
46 for (size_t ai = 0; ai < arm->numberOfActuators(); ai++) {
47 Actuator* actuator = arm->getActuator(ai);
48 actuator->setBus(getBus(armature.name + "." + armature.joints[ai].name,
49 (armature.joints[ai].bus != "") ? armature.joints[ai].bus : armature.bus));
50 if (!std::isnan(filterAlpha)) {
51 actuator->setLowPassFilter(filterAlpha);
52 }
53 }
54 fArmature.push_back(arm);
55 }
56 for (size_t i = 0; i < config.sensors.size(); i++) {
57 const pd::config::Sensor &sensorConfig = config.sensors[i];
58 Sensor* sensor = SensorProvider::find(sensorConfig.source, sensorConfig.name, sensorConfig.settings, sensorConfig.id);
59 if (sensor != nullptr) {
60 fSensor.push_back(sensor);
61 if (sensorConfig.hasBus()) {
62 auto bus = getBus(sensorConfig.bus);
63 if (bus != nullptr) {
64 PDLOG_INFO("[%s] on bus: %s\n",
65 sensorConfig.name.c_str(), sensorConfig.bus.c_str());
66 sensor->setBus(bus);
67 } else {
68 PDLOG_ERROR("[%s] UNRESOLVED SENSOR BUS: %s\n",
69 sensorConfig.name.c_str(), sensorConfig.bus.c_str());
70 }
71 }
72 } else {
73 PDLOG_ERROR("Unknown sensor source %s\n", sensorConfig.source.c_str());
74 }
75 }
76 for (auto armature : fArmature) {
77 for (size_t i = 0; i < armature->numberOfActuators(); i++) {
78 auto actuator = armature->getActuator(i);
79 auto bus = actuator->getBus();
80 if (bus) {
81 PDLOG_INFO("%d: Adding %s to bus %s\n", i, actuator->getName().c_str(), bus->getName().c_str());
82 fByBus[bus].push_back(actuator);
83 bus->flushUnread();
84 if (std::find(fOrderedBuses.begin(), fOrderedBuses.end(), bus) == fOrderedBuses.end()) {
85 fOrderedBuses.push_back(bus);
86 }
87 }
88 }
89 }
90 PDLOG_INFO("-------------------\n");
91 for (auto* bus : fBus) {
92 auto &vec = fByBus[bus];
93 for (auto* act : vec) {
94 PDLOG_INFO("[%s] %s\n", bus->getName().c_str(), act->getName().c_str());
95 }
96 }
97
98 }

167 void report(Stream output, pd::ControllerEvent* controllerEvent) {
168 printStream(output, "{");
169 if (controllerEvent != nullptr && controllerEvent->fValid) {
170 printStream(output, "\"event\":{");
171 printStream(output, "\"values\":[");
172 for (unsigned i = 0; i < SizeOfArray(controllerEvent->fValues); i++) {
173 printStream(output, "%.2f%s", controllerEvent->fValues[i], (i+1<SizeOfArray(controllerEvent->fValues)) ? ",": "");
174 }
175 printStream(output, "],");
176 printStream(output, "\"button\":%d,", controllerEvent->fButtonPressed);
177 printStream(output, "\"failsafe\":%d,", controllerEvent->fFailSafe);
178 printStream(output, "\"lostframe\":%d", controllerEvent->fLostFrame);
179 printStream(output, "},");
180 }
181 if (isPolicyRunning()) {
182 printStream(output, "\"policy\":{");
183 printStream(output, "\"name\":\"%s\",", fPolicy.c_str());
184 printStream(output, "\"dt\":%f", fPolicyDT);
185 printStream(output, "},");
186 }
187 printStream(output, "\"motors\":[");
188 bool needComma = false;
189 for (size_t ai = 0; ai < fArmature.size(); ai++) {
190 Armature& armature = *fArmature[ai];
191 for (size_t i = 0; i < armature.numberOfActuators(); i++) {
192 Actuator& actuator = armature[i];
193 if (actuator.shouldIgnore())
194 continue;
195 if (needComma)
196 printStream(output, ",");
197 printStream(output, "{");
198 printStream(output, "\"name\": \"%s\",", actuator.getName().c_str());
199 printStream(output, "\"error\":%d", actuator.hasError() ? 1 : 0);
200 auto pos = actuator.getPosition();
201 if (pos == pos)
202 printStream(output, ",\"pos\":%.4f", pos);
203 auto rad = actuator.getJointOffset();
204 if (rad == rad)
205 printStream(output, ",\"rad\":%.4f", rad);
206 auto angle = actuator.getDegrees();
207 if (angle == angle)
208 printStream(output, ",\"raw\":%.4f", angle);
209 auto kp = actuator.getKP();
210 if (kp == kp)
211 printStream(output, ",\"kp\":%.4f", kp);
212 auto kd = actuator.getKD();
213 if (kd == kd)
214 printStream(output, ",\"kd\":%.4f", kd);
215 switch (actuator.getType()) {
216 case pd::Go1:
217 case pd::Go2:
218 case pd::UkiA1:
219 printStream(output, ",\"tau\":%.4f,", actuator.getFeedbackTorque());
220 printStream(output, "\"vel\":%.4f", actuator.getFeedbackVelocity());
221 break;
222 case pd::CyberGear:
223 case pd::Dynamixel:
224 // Add more feedback
225 printStream(output, ",\"tau\":%.4f,", actuator.getFeedbackTorque());
226 printStream(output, "\"vel\":%.4f", actuator.getFeedbackVelocity());
227 //printStream(output, "\"temp\":%.4f,", actuator.getFeedbackTemp());
228 // printStream(output, "\"force\":%.1f", 0.0);
229 break;
230 case pd::Servo:
231 default:
232 break;
233 }
234 printStream(output, "}");
235 needComma = true;
236 }
237 }
238 printStream(output, "],");
239 printStream(output, "\"sensors\":[");
240 Sensor::Event event;
241 needComma = false;
242 for (auto sensor : fSensor) {
243 if (!sensor->currentEvent(event))
244 continue;
245 if (needComma)
246 printStream(output, ",");
247 printStream(output, "{");
248 printStream(output, "\"name\": \"%s\",", sensor->getName().c_str());
249 switch (event.type) {
250 case Sensor::GYROSCOPE:
251 printStream(output, "\"gyro\": {");
252 printStream(output, "\"x\": %.2f,", event.gyro.x);
253 printStream(output, "\"y\": %.2f,", event.gyro.y);
254 printStream(output, "\"z\": %.2f,", event.gyro.z);
255 printStream(output, "\"accuracy\": %d", event.gyro.accuracy);
256 printStream(output, "}");
257 break;
258 case Sensor::INERTIAL:
259 printStream(output, "\"inertial\": {");
260 printStream(output, "\"roll\": %.2f,", event.inertial.roll);
261 printStream(output, "\"pitch\": %.2f,", event.inertial.pitch);
262 printStream(output, "\"heading\": %.2f,", event.inertial.heading);
263 printStream(output, "\"i\": %.2f,", event.inertial.quat.x);
264 printStream(output, "\"j\": %.2f,", event.inertial.quat.y);
265 printStream(output, "\"k\": %.2f,", event.inertial.quat.z);
266 printStream(output, "\"r\": %.2f,", event.inertial.quat.w);
267 printStream(output, "\"x\": %.2f,", event.inertial.gyro.x);
268 printStream(output, "\"y\": %.2f,", event.inertial.gyro.y);
269 printStream(output, "\"z\": %.2f,", event.inertial.gyro.z);
270 printStream(output, "\"ax\": %.2f,", event.inertial.acc.x);
271 printStream(output, "\"ay\": %.2f,", event.inertial.acc.y);
272 printStream(output, "\"az\": %.2f,", event.inertial.acc.z);
273 printStream(output, "\"accuracy\": %d",
274 std::min(event.inertial.gyro.accuracy, event.inertial.acc.accuracy));
275 printStream(output, "}");
276 break;
277 case Sensor::CURRENT:
278 printStream(output, "\"current\": %.1f", event.current);
279 break;
280 case Sensor::BMS:
281 printStream(output, "\"voltage\": %.1f,", event.bms.voltage);
282 printStream(output, "\"current\": %.1f,", event.bms.current);
283 printStream(output, "\"stateOfCharge\": %.1f", event.bms.stateOfCharge);
284 break;
285 case Sensor::VOLTAGE:
286 printStream(output, "\"voltage\": %.1f", event.voltage);
287 break;
288 default:
289 break;
290 }
291 printStream(output, "}");
292 needComma = true;
293 }
294 printStream(output, "]}\r\n");
295 }

549 bool init() {
550 bool success = true;
551 for (size_t armi = 0; armi < fArmature.size(); armi++) {
552 Armature& armature = *fArmature[armi];
553 for (size_t i = 0; i < armature.numberOfActuators(); i++) {
554 Actuator& actuator = armature[i];
555 pd::Bus* bus = actuator.getBus();
556 if (!actuator.update()) {
557 PDLOG_ERROR("[%s] MOTOR NOT RESPONDING: %s.%s\n",
558 bus->getName().c_str(),
559 armature.getName().c_str(),
560 actuator.getName().c_str());
561 bus->setError(true);
562 success = false;
563 }
564 if (!actuator.isResponding()) {
565 actuator.setIgnore();
566 }
567 }
568 }
569 return success;
570 }

588 clearBusError();
589
590 uint64_t now = pd::platform::currentTimeMillis();
591
592 bool ok = true;
593 for (auto arm : fArmature) {
594 ok &= arm->prepareAll(now);

596 if (!ok) {
597 // Async update not supported fall back on sync update

599 return false;
600 }
601 size_t maxStripes = 0;
602 for (auto* bus : fOrderedBuses)
603 maxStripes = std::max(maxStripes, fByBus[bus].size());
604
605 // PDLOG_INFO("============================maxStripes=%d\n", maxStripes);
606 for (size_t stripe = 0; stripe < maxStripes; ++stripe) {
607 PDLOG_INFO("STRIPE: %d\n", stripe);
608 for (auto* bus : fOrderedBuses) {
609 auto &vec = fByBus[bus];
610 // PDLOG_INFO("[%ld] bus=%s\n", stripe, bus->getName().c_str());
611 if (stripe < vec.size()) {
612 // for (size_t i = 0; i < stripe; i++) {
