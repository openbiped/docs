28 if (!failsafe && channel >= 0 && channel < NUM_CHANNELS) {

43 bool read(Packet& packet) {
44 bool haveData = false;
45 if (fBus != nullptr) {
46 // Greedly consume all data
47 do {
48 if (parse(packet)) {
49 haveData = true;
50 }
51 } while (fBus->available());
52 }
53 return haveData;
54 }

56 bool write(const Packet& packet) {

105 static constexpr uint8_t FOOTER2 = 0x04;
106 static constexpr uint8_t CH17_MASK = 0x01;
107 static constexpr uint8_t CH18_MASK = 0x02;
108 static constexpr uint8_t LOST_FRAME_MASK = 0x04;
109 static constexpr uint8_t FAILSAFE_MASK = 0x08;

111 bool parse(Packet& packet) {
112 auto isHeaderValid = [&](uint8_t cur, uint8_t prev) -> bool {
113 return (cur == HEADER) && ((prev == FOOTER) || ((prev & 0x0F) == FOOTER2));
114 };
115
116 auto isFooterValid = [&](uint8_t byte) -> bool {
117 return (byte == FOOTER) || ((byte & 0x0F) == FOOTER2);
118 };
119
120 while (fBus->available()) {
121 fCurByte = fBus->read();
122
123 if (fState == 0) {
124 // Waiting for a header byte.
125 if (isHeaderValid(fCurByte, fPrevByte)) {
126 fBuf[fState++] = fCurByte;
127 } else {
128 fState = 0;
129 }
130 } else if (fState < PAYLOAD_LEN + HEADER_LEN) {
131 // Collect payload bytes.
132 fBuf[fState++] = fCurByte;
133 } else if (fState < PAYLOAD_LEN + HEADER_LEN + FOOTER_LEN) {
134 // Now in the footer section.
135 // Save the current byte as the previous byte for the next iteration.
136 fPrevByte = fCurByte;
137
138 // Reset the state for the next message regardless of success/failure.
139 fState = 0;
140
141 if (isFooterValid(fCurByte)) {
142 // Decode the channel data from the collected buffer.
143 packet.ch[0] = int16_t(fBuf[1] | ((fBuf[2] << 8) & 0x07FF));
144 packet.ch[1] = int16_t((fBuf[2] >> 3) | ((fBuf[3] << 5) & 0x07FF));
145 packet.ch[2] = int16_t((fBuf[3] >> 6) | (fBuf[4] << 2) | ((fBuf[5] << 10) & 0x07FF));
146 packet.ch[3] = int16_t((fBuf[5] >> 1) | ((fBuf[6] << 7) & 0x07FF));
147 packet.ch[4] = int16_t((fBuf[6] >> 4) | ((fBuf[7] << 4) & 0x07FF));
148 packet.ch[5] = int16_t((fBuf[7] >> 7) | (fBuf[8] << 1) | ((fBuf[9] << 9) & 0x07FF));
149 packet.ch[6] = int16_t((fBuf[9] >> 2) | ((fBuf[10] << 6) & 0x07FF));
150 packet.ch[7] = int16_t((fBuf[10] >> 5) | ((fBuf[11] << 3) & 0x07FF));
151 packet.ch[8] = int16_t(fBuf[12] | ((fBuf[13] << 8) & 0x07FF));
152 packet.ch[9] = int16_t((fBuf[13] >> 3) | ((fBuf[14] << 5) & 0x07FF));
153 packet.ch[10] = int16_t((fBuf[14] >> 6) | (fBuf[15] << 2) | ((fBuf[16] << 10) & 0x07FF));
154 packet.ch[11] = int16_t((fBuf[16] >> 1) | ((fBuf[17] << 7) & 0x07FF));
155 packet.ch[12] = int16_t((fBuf[17] >> 4) | ((fBuf[18] << 4) & 0x07FF));
156 packet.ch[13] = int16_t((fBuf[18] >> 7) | (fBuf[19] << 1) | ((fBuf[20] << 9) & 0x07FF));
157 packet.ch[14] = int16_t((fBuf[20] >> 2) | ((fBuf[21] << 6) & 0x07FF));
158 packet.ch[15] = int16_t((fBuf[21] >> 5) | ((fBuf[22] << 3) & 0x07FF));
159
160 // Decode extra channel data and status flags from fBuf[23].
161 auto statusFlags = fBuf[23];
162 packet.ch17 = ((statusFlags & CH17_MASK) != 0);
163 packet.ch18 = ((statusFlags & CH18_MASK) != 0);
164 packet.lost_frame = ((statusFlags & LOST_FRAME_MASK) != 0);
165 packet.failsafe = ((statusFlags & FAILSAFE_MASK) != 0);
166 return true;
167 } else {
168 return false;
169 }
170 } else {
171 // Should never get here; reset state if it does.
172 fState = 0;
173 }
174
175 // Save current byte for the next loop iteration.
176 fPrevByte = fCurByte;
177 }
178 return false;
179 }
180};
