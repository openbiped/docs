187// Little-endian: Swap bytes.
188inline uint32_t hton_32(uint32_t hostlong) {
189 return ((hostlong & 0x000000FFU) << 24) |
190 ((hostlong & 0x0000FF00U) << 8) |
191 ((hostlong & 0x00FF0000U) >> 8) |
192 ((hostlong & 0xFF000000U) >> 24);
193}

195inline uint32_t ntoh_32(uint32_t netlong) {
196 // The conversion is symmetric.

198 ((netlong & 0x0000FF00U) << 8) |
199 ((netlong & 0x00FF0000U) >> 8) |
200 ((netlong & 0xFF000000U) >> 24);
