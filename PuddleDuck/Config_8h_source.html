100 * Default Values:

102 * providing a balanced initial response to error without being too aggressive.
103 */

105
106 /*
107 * - Default kd = 0.01: This provides a small amount of damping, which can help to
108 * smooth out the control response without being too restrictive.
109 */
110 static constexpr double DEFAULT_KD = 0.01;
111
112 /*
113 * - Default ki = 0: This is the default value for the integral gain, as integral
114 * control is not always necessary and can introduce stability issues if not
115 * tuned correctly.

117 static constexpr double DEFAULT_KI = 0;

120 pd::String name;

315template<>
316struct convert<pd::config::Range> {
317 static pd::String encodeWithPrecision(double value, int precision) {
318 std::stringstream ss;
319 if (std::isnan(value)) {
320 ss << ".nan";
321 } else if (std::isinf(value)) {
322 if (std::signbit(value)) {
323 ss << "-.inf";
324 } else {
325 ss << ".inf";
326 }
327 } else {
328 ss << std::fixed << std::setprecision(precision) << value;
329 }
330 return ss.str();
331 }
332
333 static Node encode(const pd::config::Range& rhs) {
334 Node node;
335
336 node.push_back(encodeWithPrecision(rhs.value[0], 4));
337 node.push_back(encodeWithPrecision(rhs.value[1], 4));
338 node.SetStyle(YAML::EmitterStyle::Flow);
339 return node;
340 }
341
342 static bool decode(const Node& node, pd::config::Range& rhs) {
343 if (!node.IsSequence() || node.size() != 2) {
344 return false;
345 }
346 rhs.value[0] = node[0].as<double>();
347 rhs.value[1] = node[1].as<double>();
348 return true;
349 }
350};

363 if (rhs.kp != rhs.DEFAULT_KP)
364 node["kp"] = rhs.kp;
365 if (rhs.kd != rhs.DEFAULT_KD)
366 node["kd"] = rhs.kd;
367 if (rhs.ki != rhs.DEFAULT_KI)
368 node["ki"] = rhs.ki;
369 if (rhs.tau != rhs.DEFAULT_TAU)
370 node["tau"] = rhs.tau;
371 if (rhs.invert)
372 node["invert"] = rhs.invert;
373 if (rhs.optional)
374 node["optional"] = rhs.optional;
375 return node;
376 }
377
378 static bool decode(const Node& node, pd::config::Joint& rhs) {
379 if (!node["name"] ||
380 !node["type"] ||
381 !node["id"])
382 {
383 return false;
384 }
385 pd::String type = node["type"].as<pd::String>();
386 rhs.type = rhs.getTypeFromName(type);
387 if (rhs.type == pd::UnknownMotor) {
388 return false;
389 }
390 rhs.name = node["name"].as<pd::String>();
391 if (node["bus"])
392 rhs.bus = node["bus"].as<pd::String>();
393 rhs.id = node["id"].as<int>();
394 if (node["range"])
395 rhs.range = node["range"].as<pd::config::Range>();
396 else
397 rhs.range.setInvalid();
398 rhs.zero = (node["zero"]) ?
399 node["zero"].as<double>() : 0;
400 rhs.start = (node["start"]) ?
401 node["start"].as<double>() : 0;
402 if (node["kp"])
403 rhs.kp = node["kp"].as<double>();
404 else

406 if (node["ki"])

412 else
413 rhs.kd = rhs.DEFAULT_KD;
414 if (node["tau"])
415 rhs.tau = node["tau"].as<double>();
416 else
417 rhs.tau = rhs.DEFAULT_TAU;
418 if (node["invert"])

420 else
421 rhs.invert = false;
422 if (node["optional"])
423 rhs.optional = node["optional"].as<bool>();
424 else
425 rhs.optional = false;

479template<>
480struct convert<pd::config::Robot> {
481 static Node encode(const pd::config::Robot& rhs) {
482 Node node;
483 node["frequency"] = rhs.frequency;
484 node["cutoff"] = rhs.cutoff;
485 node["bus"] = YAML::Node(YAML::NodeType::Sequence);
486 for (int i = 0; i < rhs.bus.size(); ++i) {
487 node["bus"].push_back(rhs.bus[i]);
488 }
489 node["armature"] = YAML::Node(YAML::NodeType::Sequence);
490 for (int i = 0; i < rhs.armature.size(); ++i) {
491 node["armature"].push_back(rhs.armature[i]);
492 }
493 node["sensor"] = YAML::Node(YAML::NodeType::Sequence);
494 for (int i = 0; i < rhs.sensors.size(); ++i) {
495 node["sensor"].push_back(rhs.sensors[i]);
496 }
497 return node;
498 }
499
500 static bool decode(const Node& node, pd::config::Robot& rhs) {
501 if (node["frequency"]) {
502 rhs.frequency = node["frequency"].as<double>();
503 }
504 if (node["cutoff"]) {
505 rhs.cutoff = node["cutoff"].as<double>();
506 }
507 if (!node["bus"] ||
508 !node["armature"])
509 {
510 return false;
511 }
512 for (auto it = node["bus"].begin(); it != node["bus"].end(); it++) {
513 rhs.bus.push_back(it->as<pd::config::Bus>());
514 }
515 for (auto it = node["armature"].begin(); it != node["armature"].end(); it++) {
516 rhs.armature.push_back(it->as<pd::config::Armature>());
517 }
518 for (auto it = node["sensor"].begin(); it != node["sensor"].end(); it++) {
519 rhs.sensors.push_back(it->as<pd::config::Sensor>());
520 }
521 return true;
522 }
