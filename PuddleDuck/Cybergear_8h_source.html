27 virtual bool update() override {
28 if (shouldIgnore()) {
29 return true;
30 }
31 if (fBus == nullptr) {
32 PDLOG_ERROR("UNRESOLVED ACTUATOR BUS\n");
33 return false;
34 }
35 switch (fRunMode) {
36 case MODE_UNKNOWN:
37 if (initialize() && requestStatus()) {
38 printf("requestStatus\r\n");
39 return process();
40 }
41 return false;
42 case MODE_MOTION:
43 if (fActive) {
44 if (pd::Log::isVerboseMove()) {
45 PDLOG_INFO("[%s] %f\n", getName().c_str(), fPosNow);
46 }
47 if (!control(degreesToRadians(fPosNow), 0, fKP, fKD, fTau))
48 return false;
49 } else if (!control(0, 0, 0, 0, 0)) {
50 return false;
51 }
52 return process();
53 case MODE_POSITION:
54 PDLOG_ERROR("NYI MODE_POSITION\n");
55 return false;
56 case MODE_SPEED:
57 PDLOG_ERROR("NYI MODE_SPEED\n");
58 return false;
59 case MODE_CURRENT:
60 PDLOG_ERROR("NYI MODE_CURRENT\n");
61 return false;
62 }
63 return false;
64 }

233 bool control(float pos, float speed, float kp, float kd, float tau) {
234 uint16_t p = float_to_uint(pos, P_MIN, P_MAX, 16);
235 uint16_t v = float_to_uint(speed, V_MIN, V_MAX, 16);
236 uint16_t kpi = float_to_uint(kp, KP_MIN, KP_MAX, 16);
237 uint16_t kdi = float_to_uint(kd, KD_MIN, KD_MAX, 16);
238
239 uint8_t data[] = {
240 uint8_t(p >> 8),
241 uint8_t(p & 0xFF),
242
243 uint8_t(v >> 8),
244 uint8_t(v & 0xFF),
245
246 uint8_t(kpi >> 8),
247 uint8_t(kpi & 0xFF),
248
249 uint8_t(kdi >> 8),
250 uint8_t(kdi & 0xFF)
251 };
252 uint16_t torque = float_to_uint(tau, T_MIN, T_MAX, 16);
253 return send(CMD_POSITION, torque, 8, data);
254 }

313 bool send(uint8_t cmd_id, uint16_t option, uint8_t len, uint8_t* data) {
314 uint32_t id = (0x00000000 | cmd_id << 24 | option << 8 | fMotorID);
315 // printf("ID=%08uX fMotorID=%d\r\n", id, fMotorID);
316 if (fBus->write(data, len, id, true) == len) {
317 return true;
318 }
319 // dbg("failed to send");
320 return false;
321 }

329 bool write_float_data(Address addr, float value, float min, float max) {
330 uint8_t data[8] = {0};
331 data[0] = addr & 0x00FF;
332 data[1] = addr >> 8;
333
334 float val = (max < value) ? max : value;
335 val = (min > value) ? min : value;
336 memcpy(&data[4], &val, 4);
337 return send(CMD_RAM_WRITE, getBusID(), 8, data);
338 }

340 int float_to_uint(float x, float x_min, float x_max, int bits) {
341 float span = x_max - x_min;
342 float offset = x_min;
343 if (x > x_max)
344 x = x_max;
345 else if (x < x_min)
346 x = x_min;
347 return (int) ((x - offset) * ((float)((1 << bits) - 1)) / span);
348 }
349
350 float uint_to_float(uint16_t x, float x_min, float x_max) {
351 uint16_t type_max = 0xFFFF;
352 float span = x_max - x_min;
353 return (float) x / type_max * span + x_min;
354 }

356 int receive_motor_data(MotorStatus &mot)
357 {
358 uint32_t id;
359 uint8_t data[8] = {0};
360 if (fBus->read(data, sizeof(data), &id) != sizeof(data)) {
361 return 0;
362 }
363 // if id is not mine
364 uint8_t receive_can_id = (id & 0xFF);
365 uint8_t motor_can_id = ((id >> 8) & 0xFF);
366 if (receive_can_id != getBusID() ||
367 motor_can_id != fMotorID)
368 {
369 return -1;
370 }
371
372 // check packet type
373 uint8_t packet_type = ((id >> 24) & 0x3F);
374 if (packet_type != CMD_RESPONSE) {
375 return -1;
376 }
377
378 mot.raw_position = data[1] | data[0] << 8;
379 mot.raw_velocity = data[3] | data[2] << 8;
380 mot.raw_torque = data[5] | data[4] << 8;
381 mot.raw_temperature = data[7] | data[6] << 8;
382
383 // convert motor data
384 mot.motor_id = motor_can_id;
385 mot.position = uint_to_float(mot.raw_position, P_MIN, P_MAX);
386 mot.velocity = uint_to_float(mot.raw_velocity, V_MIN, V_MAX);
387 mot.torque = uint_to_float(mot.raw_torque, T_MIN, T_MAX);
388 mot.temperature = mot.raw_temperature;
389
390 mot.hasUnderVoltage = (id >> 16) & 1;
391 mot.hasOverCurrent = (id >> 17) & 1;
392 mot.hasOverTemperature = (id >> 18) & 1;
393 mot.hasMagneticEncodingError = (id >> 19) & 1;
394 mot.hasHallEncoderError = (id >> 20) & 1;
395 mot.hasCalibrationError = (id >> 21) & 1;
396 mot.mode = RunMode((id >> 22) & 3);
397
398 setFeedback(
399 radiansToDegrees(mot.position / GEAR_RATIO),
400 mot.velocity,
401 mot.torque,
402 mot.temperature);
403 if (mot.hasUnderVoltage)
404 setError(Error::VoltageError);
405 else if (mot.hasOverCurrent)
406 setError(Error::CurrentError);
407 else if (mot.hasOverTemperature)
408 setError(Error::TemperatureError);
409 else if (mot.hasMagneticEncodingError)
410 setError(Error::EncoderError);
411 else if (mot.hasHallEncoderError)
412 setError(Error::EncoderError);
413 else if (mot.hasCalibrationError)
414 setError(Error::EncoderError);
415 else
416 clearError();
417 return 1;
418 }
