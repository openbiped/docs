255class SensorProvider {
256public:
257 SensorProvider(pd::String name) :
258 fName(name)
259 {
260 if (*head() == NULL)
261 *head() = this;
262 if (*tail() != NULL)
263 (*tail())->fNext = this;
264 *tail() = this;
265 }
266
267 static Sensor* find(pd::String source, pd::String name, pd::String settings, int32_t id = 0) {
268 for (auto provider = *head(); provider != nullptr; provider = provider->fNext) {
269 if (provider->fName == source) {
270 return provider->create(name, settings, id);
271 }
272 }
273 PDLOG_ERROR("SENSOR NOT FOUND: %s\n", source.c_str());
274 return nullptr;
275 }
276
277 inline pd::String getName() const {
278 return fName;
279 }
280
281protected:
282 pd::String fName;
283
284 virtual Sensor* create(pd::String name, pd::String settings, int32_t id) = 0;
285
286private:
287 SensorProvider* fNext = nullptr;
288
289 static SensorProvider** head() {
290 static SensorProvider* sHead;
291 return &sHead;
292 }
293
294 static SensorProvider** tail() {
295 static SensorProvider* sTail;
296 return &sTail;
297 }
