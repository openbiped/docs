24 float sbus_to_float(int sbus_value) {
25 // 1) Define input limits and midpoint
26 const int min_input = 170;
27 const int max_input = 1800;
28 const int midpoint = (min_input + max_input) / 2; // ~985
29
30 // 2) Define deadband as a percentage of the total range
31 const float deadband_percent = 0.05f; // 5%
32 const float total_range = (float)(max_input - min_input); // e.g. 1630
33 const float deadband_range = total_range * deadband_percent / 2; // +/- around midpoint
34
35 // 3) Compute offset from midpoint (this can be negative or positive)
36 float offset = (float)sbus_value - (float)midpoint;
37
38 // 4) If within deadband, return 0.0
39 if (fabsf(offset) <= deadband_range) {
40 return 0.0f;
41 }
42
43 // 5) "Effective" half-range is total_range/2 minus deadband_range.
44 // i.e. the portion from midpoint to min/max not included in deadband.
45 float half_range = (total_range / 2.0f) - deadband_range;
46
47 // 6) Figure out sign and how far we are outside deadband
48 float sign = (offset > 0.0f) ? +1.0f : -1.0f;
49 float magnitude = fabsf(offset) - deadband_range;
50 float normalized = magnitude / half_range; // un-clamped
51 float final_value = sign * normalized; // negative if below midpoint, positive if above
52
53 // 7) Clamp to [-1, +1] just to be safe
54 if (final_value > 1.0f) final_value = 1.0f;
55 if (final_value < -1.0f) final_value = -1.0f;
56
57 return final_value;
58 }
