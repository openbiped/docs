144 void info_throttled(const char* id, const char* msg) {
145 time_t now = time(NULL);
146 // Use an unordered_map to track last print time by error id string.
147 static std::unordered_map<std::string, time_t> last_printed;
148 std::string key(id);
149 if (last_printed.find(key) != last_printed.end() &&
150 (now - last_printed[key]) < 1) {
151 return; // Throttle: less than one second since last print for this id.
152 }
153 last_printed[key] = now;
154
155 if (fBytewise) {
156 fBytewise = false;
157 info("\n");
158 }
159 // Combine the id and message for logging.
160 info("%s: %s", id, msg);
161 }
162
163 // New function: error_throttled
164 // Accepts two fully formatted C strings:
165 // - id: an identifier string (e.g. "DXL3")
166 // - msg: the error message string (e.g. "Some error message 42")
167 // The function prints the error (using error()) only if at least one second has passed
168 // since the last print for the given id.
169 void error_throttled(const char* id, const char* msg) {
170 time_t now = time(NULL);
171 // Use an unordered_map to track last print time by error id string.
172 static std::unordered_map<std::string, time_t> last_printed;
173 std::string key(id);
174 if (last_printed.find(key) != last_printed.end() &&
175 (now - last_printed[key]) < 1) {
176 return; // Throttle: less than one second since last print for this id.
177 }
178 last_printed[key] = now;
179
180 if (fBytewise) {
181 fBytewise = false;
182 error("\n");
183 }
184 // Combine the id and message for logging.
185 error("%s: %s", id, msg);
186 }
