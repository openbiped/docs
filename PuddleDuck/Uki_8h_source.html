77 /*
78 * setTau(float tau):
79 * - Description: Sets the torque (tau) for the motor. The torque represents the rotational force being applied.
80 * - Parameters:
81 * - tau: The desired torque value in Newton-meters (Nm).
82 * - Usage: Use this function to control the amount of rotational force the motor applies.
83 */
84 inline void setTau(float tau) {
85 tau = pd::max(pd::min(tau, 127), -127);
86 int16_t tau_int = int16_t(tau*256);
87 cmd.fTau[0] = uint8_t((tau_int>>0)&0xFF);
88 cmd.fTau[1] = uint8_t((tau_int>>8)&0xFF);
89 }

91 /*
92 * setDQ(float dq):
93 * - Description: Sets the angular velocity (dq) for the motor. The angular velocity is the rate of change of the motor's position.
94 * - Parameters:
95 * - dq: The desired angular velocity in radians per second (rad/s).
96 * - Usage: Use this function to control the speed at which the motor rotates.
97 */
98 inline void setDQ(float dq) {
99 dq *= GEAR_RATIO;
100 dq = pd::max(pd::min(dq, 127.0), -127.0);
101 int16_t dq_int = int16_t(dq*128);
102 cmd.fDQ[0] = uint8_t((dq_int>>0)&0xFF);
103 cmd.fDQ[1] = uint8_t((dq_int>>8)&0xFF);

115 /*
116 * setQ(float q):
117 * - Description: Sets the position (q) for the motor. The position represents the angular displacement of the motor shaft.
118 * - Parameters:
119 * - q: The desired position in radians * gear shaft ratio.
120 * - Usage: Use this function to control the angular position of the motor shaft.
121 */
122 inline void setQ(float q) {
123 q = pd::max(pd::min(q, 411774), -411774);
124 int32_t q_int = int32_t(q * 16384.0 / 6.2832);
125 cmd.fQ[0] = uint8_t((q_int>>0)&0xFF);
126 cmd.fQ[1] = uint8_t((q_int>>8)&0xFF);
127 cmd.fQ[2] = uint8_t((q_int>>16)&0xFF);
128 cmd.fQ[3] = uint8_t((q_int>>24)&0xFF);
129 }
130
131 /*
132 * setQRadians(float q):
133 * - Description: Sets the position for the motor in radians.
134 * - Parameters:
135 * - q: The desired position in radians.
136 * - Usage: Use this function to control the angular position of the motor shaft.
137 */
138 inline void setQRadians(float radians) {
139 setQ(radians * GEAR_RATIO);
140 }

145 * - Parameters:

153 /*
154 * Proportional Gain (kp):
155 * - Definition: The proportional gain kp determines the response of the controller to the current error.
156 * It is a scaling factor applied to the error value.
157 * - Effect:
158 * - Increasing kp: This results in a stronger response to the error, making the system react more quickly
159 * to discrepancies between the desired and actual positions (or other controlled variables). However, if kp
160 * is too high, it can cause the system to become overly aggressive, leading to oscillations or instability.
161 * - Decreasing kp: This results in a weaker response to the error, making the system slower to react. While this
162 * can reduce oscillations and improve stability, it may also result in sluggish performance and a slower return
163 * to the desired position.
164 */
165 inline void setKP(float output_kp) {
166 auto rotor_kp = (output_kp / (GEAR_RATIO * GEAR_RATIO)) / 26.07;
167 int16_t kp_int = int16_t(rotor_kp*2048);
168 cmd.fKP[0] = uint8_t((kp_int>>0)&0xFF);
169 cmd.fKP[1] = uint8_t((kp_int>>8)&0xFF);
170 }
171
172 /*
173 * Derivative Gain (kd):
174 * - Definition: The derivative gain kd determines the response of the controller to the rate of change of the error.
175 * It is a scaling factor applied to the derivative (rate of change) of the error.
176 * - Effect:
177 * - Increasing kd: This results in a stronger damping effect, helping to smooth out the response and reduce overshoot
178 * and oscillations. Essentially, it predicts future error based on the current rate of change and adjusts the control
179 * effort accordingly.
180 * - Decreasing kd: This results in a weaker damping effect, which can lead to increased oscillations and overshoot if
181 * the proportional gain kp is high. However, if kd is too low, the system might not sufficiently counteract rapid
182 * changes in the error, leading to a less smooth response.
183 */
184 inline void setKD(float output_kd) {
185 auto rotor_kd = (output_kd / (GEAR_RATIO * GEAR_RATIO)) * 100.0;
186 int16_t kd_int = (cmdMotorType == UkiA1) ?
187 int16_t(rotor_kd*1024) : int16_t(rotor_kd*512);
188 cmd.fKD[0] = uint8_t((kd_int>>0)&0xFF);
189 cmd.fKD[1] = uint8_t((kd_int>>8)&0xFF);
190 }

200 bool write(pd::Bus* bus, const CRC& motorCRC) {
201 uint32_t crc = motorCRC.crcWrite(&cmd, sizeof(cmd));
202 fCRC[0] = uint8_t(crc & 0xFF);
203 fCRC[1] = uint8_t(crc >> 8);
204 fCRC[2] = uint8_t((crc >> 16) & 0xFF);
205 fCRC[3] = uint8_t((crc >> 24) & 0xFF);
206 // Delay for 20 microseconds
207 pd::platform::usleep(20);
208 if (bus != nullptr && bus->write(fBytes, sizeof(fBytes)) == sizeof(fBytes)) {
209 return true;
210 }
211 return false;

324 bool read(pd::Bus* bus, const CRC& motorCRC) {
325 ssize_t numBytes = 0;
326 if (bus != nullptr && (numBytes = bus->read(fBytes, sizeof(fBytes))) == sizeof(fBytes)) {
327 // Drain the read buffer
328 //bus->flushUnread();
329 uint32_t crc = motorCRC.crcRead(&cmd, sizeof(cmd));
330 if (fCRC[0] == uint8_t(crc & 0xFF) && fCRC[1] == uint8_t(crc >> 8) &&
331 fCRC[2] == uint8_t(crc >> 16) && fCRC[3] == uint8_t(crc >> 24))
332 {
333 return true;
334 } else {
335 PDLOG_ERROR("BAD CRC EXPECTED %02X:%02X:%02X:%02X GOT %02X:%02X:%02X:%02X\n",
336 fCRC[0], fCRC[1], fCRC[2], fCRC[3],
337 uint8_t(crc & 0xFF), uint8_t(crc >> 8),
338 uint8_t(crc >> 16), uint8_t(crc >> 24));
339 for (unsigned i = 0; i < sizeof(fBytes); i++) {
340 PDLOG_ERROR("%02X ", fBytes[i]);
341 }
342 PDLOG_ERROR("\n");
343 }
344 }
345 PDLOG_ERROR("numBytes read: %d (expected %d)\n", numBytes, sizeof(fBytes));
346 invalid();
347 return false;
348 }

414 void update(Cmd& cmd, Feedback& feedback) {
415 if (fIgnore) {
416 cmd.setInvalid();
417 return;
418 }
419 move(pd::platform::currentTimeMillis());
420 cmd.setMotorID(fMotorID);
421 cmd.setDQ(0);
422 if (fActive) {
423 if (pd::Log::isVerboseMove()) {
424 if (inVelocityMode()) {
425 PDLOG_INFO("[%s] KP=%f KD=%f DQ=%f\n", getName().c_str(), fKP, fKD, fDQ);
426 } else {
427 PDLOG_INFO("[%s] KP=%f KD=%f Q=%f\n", getName().c_str(), fKP, fKD, fPosNow);
428 }
429 }
430 cmd.setFOCMode();
431 cmd.setKP(fKP);
432 cmd.setKD(fKD);
433 if (inVelocityMode()) {
434 cmd.setDQ(fDQ);
435 } else {
436 cmd.setQRadians(degreesToRadians(fPosNow));
437 }
438 cmd.setTau(fTau);
439 } else {
440 //cmd.setFOCMode();
441 cmd.setBrakeMode();
442 cmd.setKP(0.00);
443 cmd.setKD(0.00);
444 cmd.setQRadians(0);
445 cmd.setTau(0.0);
446 }
447 feedback.init();
448 }

450 bool update(Feedback& feedback) {
451 if (feedback.getMotorID() != fMotorID) {
452 PDLOG_WARNING("WRONG MOTOR GOT %d EXPECTING %d\n", feedback.getMotorID(), fMotorID);
453 return false;
454 }
455 setFeedback(
456 feedback.getCurrentAngle(),
457 feedback.getDQ(),
458 feedback.getTau(),
459 feedback.getTemperature());
460 if (feedback.getVoltageError()) {
461 PDLOG_ERROR("[%d] VOLTAGE ERROR %s.%s\n", fMotorID, fBus->getName().c_str(), getName().c_str());
462 setError(Error::VoltageError);
463 } else if (feedback.getCurrentError()) {
464 PDLOG_ERROR("[%d] CURRENT ERROR %s.%s\n", fMotorID, fBus->getName().c_str(), getName().c_str());
465 setError(Error::CurrentError);
466 } else if (feedback.getTemperatureError()) {
467 PDLOG_ERROR("[%d] TEMP ERROR %s.%s\n", fMotorID, fBus->getName().c_str(), getName().c_str());
468 setError(Error::TemperatureError);
469 } else if (feedback.getEncoderError()) {
470 PDLOG_ERROR("[%d] ENCODER ERROR %s.%s\n", fMotorID, fBus->getName().c_str(), getName().c_str());
471 setError(Error::EncoderError);
472 } else {
473 clearError();
474 }
475 return true;
476 }

478 virtual bool update() override {
479 if (shouldIgnore()) {
480 return true;
481 }
482 if (fBus == nullptr) {
483 PDLOG_ERROR("UNRESOLVED ACTUATOR BUS\n");
484 return false;
485 }
486 Cmd cmd;
487 Feedback feedback;
488 update(cmd, feedback);
489 if (!sendRecv(&cmd, &feedback)) {
490 fMissCount++;
491 return false;
492 }
493 return update(feedback);
494 }
495
496 bool prepare(uint64_t now) override {
497 move(now);
498 fCmd = {};
499 if (fIgnore) {
500 fCmd.setInvalid();
501 return true;
502 }
503 fCmd.setMotorID(fMotorID);
504 if (fActive) {
505 fCmd.setFOCMode();
506 fCmd.setKP(fKP);
507 fCmd.setKD(fKD);
508 fCmd.setTau(fTau);
509 if (inVelocityMode())
510 fCmd.setDQ(fDQ);
511 else
512 fCmd.setQRadians(degreesToRadians(fPosNow));
513 } else {
514 fCmd.setBrakeMode();
515 }
516 fFeedback.init();
517 return true;
518 }
519
520 bool transmit() override {
521 if (!fCmd.isValid())
522 return true;
523 CRC crc;
524 if (!fCmd.write(fBus, crc)) {
525 PDLOG_ERROR("FAILED TO WRITE MOTOR COMMAND TO %s\n", fBus->getName().c_str());
526 return false;
527 }
528 return true;
529 }
530
531 bool receive() override {
532 CRC crc;
533 if (!fFeedback.read(fBus, crc)) {
534 PDLOG_INFO("BAD FEEDBACK READ\n");
535 fBus->flushUnread();
536 return false;
537 }
538 setFeedback(
539 fFeedback.getCurrentAngle(),
540 fFeedback.getDQ(),
541 fFeedback.getTau(),
542 fFeedback.getTemperature());
543 return true;
544 }
545protected:
546 Cmd fCmd;
547 Feedback fFeedback;
548
549 bool sendRecv(Cmd* cmd, Feedback* feedback) {
550 feedback->init();
551 if (!cmd->isValid()) {
552 return true;
553 }
554 if (fBus == nullptr) {
555 return false;
556 }
557 CRC crc;
558 if (!cmd->write(fBus, crc)) {
559 PDLOG_ERROR("FAILED TO WRITE MOTOR COMMAND TO %s\n", fBus->getName().c_str());
560 return false;
561 }
562 if (!feedback->read(fBus, crc)) {
563 // Drain the read buffer
564 fBus->flushUnread();
565 // On power up the first Uki motor seems to often respond with garbage
566 // so lets just try one more time.
567 feedback->init();
568 if (!cmd->write(fBus, crc)) {
569 PDLOG_ERROR("FAILED TO WRITE MOTOR COMMAND TO %s\n", fBus->getName().c_str());
570 return false;
571 }
572 if (!feedback->read(fBus, crc)) {
573 // PDLOG_ERROR("[%d] CRC %s.%s\n", fMotorID, fBus->getName().c_str(), getName().c_str());
574 return false;
575 }
576 }
577 return true;
578 }
